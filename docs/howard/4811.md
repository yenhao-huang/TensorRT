## 2. dev

`plugin/resizeWithPadPlugin/`新增V3 core/build/runtime與CUDA kernel；`plugin/CMakeLists.txt`納入正常build，`plugin/api/inferPlugin.cpp`納入正式初始化。一次kernel完成resize、pad、optional channel reversal與normalization，輸出NCHW；可選transform output。支援scalar或每channel pad，channels可非3，swap要求3channels。clone重新建立serialization field pointers。

設計細節在README：nearest asymmetric floor，linear half-pixel及edge clamp；resize尺寸round-half-up且至少1；center前方floor、odd額外padding置後。metadata記錄理論等比例scale，整數resize造成的實際pixel ratio可能略有差異。`normalize`同時作用於image及padding；pad color為output channel順序。

Review修正：geometry使用double計算rounding，避免大dimension的float轉int越界；input/output volume皆檢查int64 byte-size overflow；移除不會被runtime接受的UINT8 format/kernel分支；負面測試直接檢查engine build拒絕，避免廣泛捕捉AssertionError而誤判通過。

### UINT8 限制

初稿真正向plugin送UINT8，builder報：

```text
UInt8 datatype formats are not supported. In getAllowedFormatsV3 ... pluginV3BuilderUtils.cpp:314
```

`include/NvInferRuntimeBase.h`對UINT8的說明要求network-level UINT8先轉FLOAT/HALF再給其他層。測試及README提供GPU圖內Cast，因此整張圖仍可接收UINT8且不需CPU resize/padding，但會有中間floating-point tensor，**不是直接UINT8 plugin介面**。此差異已提出確認，AC-2目前保留blocked。

### export_classifier.py

```python
from pathlib import Path
import json
import torch
from torchvision.models import resnet18, ResNet18_Weights
root=Path('/data/tenserrt/data/4811')
weights=ResNet18_Weights.DEFAULT
class Classifier(torch.nn.Module):
 def __init__(self):
  super().__init__(); self.model=resnet18(weights=weights).eval()
  self.register_buffer('mean',torch.tensor([.485,.456,.406]).reshape(1,3,1,1))
  self.register_buffer('std',torch.tensor([.229,.224,.225]).reshape(1,3,1,1))
 def forward(self,x): return self.model((x-self.mean)/self.std)
model=Classifier().eval()
torch.onnx.export(model,torch.zeros(1,3,224,224),str(root/'classifier.onnx'),input_names=['preprocessed'],output_names=['logits'],opset_version=18,dynamo=False)
(root/'categories.json').write_text(json.dumps(weights.meta['categories']))
print('exported pretrained ResNet18 + ImageNet normalization')
```

### reference_classifier.py

```python
import json
from pathlib import Path
import numpy as np
from PIL import Image
import torch
from torch.nn import functional as F
from torchvision.models import resnet18,ResNet18_Weights
root=Path('/data/tenserrt/data/4811')
x=np.asarray(Image.open('/data/tenserrt/data/4670/dog.jpg').convert('RGB'),dtype=np.float32).transpose(2,0,1)[None].copy()
np.save(root/'image.npy',x)
h,w=x.shape[2:]; scale=min(224/h,224/w); rh,rw=[int(np.floor(d*scale+.5)) for d in (h,w)]
top,left=(224-rh)//2,(224-rw)//2
resized=F.interpolate(torch.from_numpy(x),size=(rh,rw),mode='bilinear',align_corners=False)
y=F.pad(resized,(left,224-rw-left,top,224-rh-top),value=114)/255
np.save(root/'preprocessed-reference.npy',y.numpy())
mean=torch.tensor([.485,.456,.406]).reshape(1,3,1,1);std=torch.tensor([.229,.224,.225]).reshape(1,3,1,1)
model=resnet18(weights=ResNet18_Weights.DEFAULT).eval()
with torch.no_grad(): logits=model((y-mean)/std)
np.save(root/'logits-reference.npy',logits.numpy())
prob=logits.softmax(-1)[0]; k=int(prob.argmax())
result={'shape':list(x.shape),'scale':scale,'top':top,'left':left,'class':k,'label':ResNet18_Weights.DEFAULT.meta['categories'][k],'probability':float(prob[k])}
(root/'reference.json').write_text(json.dumps(result,indent=2)); print(result)
```

### run_classifier.py

```python

from pathlib import Path
import numpy as np
import onnx
from onnx import helper as h,TensorProto as T
import tensorrt as trt
sys.path.insert(0,'/workspace/tenserrt/TensorRT/samples/python')

root=Path('/data/tenserrt/data/4811'); x=np.load(root/'image.npy')
lib=ctypes.CDLL('/data/tenserrt/builds/4811/oss/libnvinfer_plugin.so')
lib.initLibNvInferPlugins.argtypes=[ctypes.c_void_p,ctypes.c_char_p]; lib.initLibNvInferPlugins.restype=ctypes.c_bool
assert lib.initLibNvInferPlugins(None,b'')
m=onnx.load(root/'classifier.onnx')
old=list(m.graph.node); del m.graph.node[:]

m.graph.node.extend(old); del m.graph.input[:]; m.graph.input.extend([h.make_tensor_value_info('image',T.FLOAT,list(x.shape))])

config=builder.create_builder_config(); config.clear_flag(trt.BuilderFlag.TF32); config.set_memory_pool_limit(trt.MemoryPoolType.WORKSPACE,512<<20)

runtime=trt.Runtime(logger); engine=runtime.deserialize_cuda_engine(plan); context=engine.create_execution_context(); mems=[];outputs={}
for i in range(engine.num_io_tensors):

 if name=='image': memcpy_host_to_device(mem.device_ptr,array)

assert context.execute_v2([m.device_ptr for m in mems])
for array,mem in outputs.values(): memcpy_device_to_host(array,mem.device_ptr)
logits=outputs['logits'][0]; reference=np.load(root/'logits-reference.npy'); y=outputs['preprocessed'][0]; refy=np.load(root/'preprocessed-reference.npy')
np.testing.assert_allclose(y,refy,atol=2e-4,rtol=2e-4)
np.testing.assert_allclose(logits,reference,atol=.01,rtol=.01)
assert int(logits.argmax())==int(reference.argmax())
result={'trt_version':trt.__version__,'class':int(logits.argmax()),'label':json.loads((root/'categories.json').read_text())[int(logits.argmax())],'logits_max_abs_error':float(np.abs(logits-reference).max()),'preprocessed_max_abs_error':float(np.abs(y-refy).max()),'transform':outputs['transform'][0].tolist()}
```
